\subsection{PUT and GET Operations}
\label{sec:operations}

% PUT Operation description
% - Discretize coordinates (raster projection)
% - Compute k hash positions using hashing trick
% - For each hash position, increment first non-maximal layer

\begin{algorithm}[ht]
\caption{PUT Operation}
\label{alg:put}
\begin{algorithmic}
\REQUIRE CGM $A = \{k, \{b_x, m_x\}\}$, point $p$
\STATE $c \leftarrow \text{RasterProjection}(p)$
\STATE $h_1, h_2 \leftarrow \hashfn(c)$
\FOR{$i = 0$ \TO $k-1$}
  \FOR{$x = 1$ \TO $\Lambda$}
    \STATE $j \leftarrow (h_1 + (i+1) \cdot h_2) \mod m_x$
    \STATE $v \leftarrow \val(j, L_{A,x})$
    \IF{$v < 2^{b_x} - 1$}
      \STATE $\val(j, L_{A,x}) \leftarrow v + 1$
      \STATE \textbf{break}
    \ENDIF
  \ENDFOR
\ENDFOR
\end{algorithmic}
\end{algorithm}

% GET Operation description
% - Returns minimum count across all k hash positions
% - Sums values across layers when overflow detected

\begin{algorithm}[ht]
\caption{GET Operation}
\label{alg:get}
\begin{algorithmic}
\REQUIRE CGM $A = \{k, \{b_x, m_x\}\}$, point $p$
\ENSURE $v_{min}$
\STATE $c \leftarrow \text{RasterProjection}(p)$
\STATE $h_1, h_2 \leftarrow \hashfn(c)$
\STATE $v_{min} \leftarrow \infty$
\FOR{$i = 0$ \TO $k-1$}
  \STATE $s \leftarrow 0$
  \FOR{$x = 1$ \TO $\Lambda$}
    \STATE $j \leftarrow (h_1 + (i+1) \cdot h_2) \mod m_x$
    \STATE $v \leftarrow \val(j, L_{A,x})$
    \STATE $s \leftarrow s + v$
    \IF{$v = 0$}
      \RETURN $0$
    \ENDIF
    \IF{$v < 2^{b_x} - 1$}
      \STATE \textbf{break}
    \ENDIF
  \ENDFOR
  \STATE $v_{min} \leftarrow \min(v_{min}, s)$
\ENDFOR
\RETURN $v_{min}$
\end{algorithmic}
\end{algorithm}
